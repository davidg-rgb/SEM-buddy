```python
"""SEM Master CLI — Google Ads campaign automation for Swedish clients."""

import typer

__version__ = "2.0.0"

def version_callback(value: bool):
    if value:
        import rich
        rich.print(f"[bold]SEM Master[/bold] v{__version__}")
        raise typer.Exit()

def create_app() -> typer.Typer:
    app = typer.Typer(
        name="sem",
        help="Google Ads campaign automation for Swedish/Nordic clients.",
        rich_markup_mode="rich",
        no_args_is_help=True,
    )

    @app.callback()
    def main_callback(
        version: bool = typer.Option(
            False, "--version", "-v",
            help="Show version and exit.",
            callback=version_callback,
            is_eager=True,
        ),
    ):
        """SEM Master — automated Google Ads campaigns for Swedish clients."""
        pass

    # Register command groups (stubs for now, implemented in plans 02-03)
    # config_app and accounts_app will be added in subsequent plans

    return app

def main():
    app = create_app()
    app()
```
